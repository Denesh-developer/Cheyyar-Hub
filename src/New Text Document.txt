rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    /* =====================================================
       HELPER FUNCTIONS
       ===================================================== */

    function isSignedIn() {
      return request.auth != null;
    }

    function isDeveloper() {
      return isSignedIn()
        && request.auth.uid == "vEIdznPIjVQiu70TwRjVKz8Dq0q2";
    }


    /* =====================================================
       USERS
       ===================================================== */

    match /users/{userId} {

      /*
        Signed-in users can read profiles.
      */
      allow read: if isSignedIn();


      /*
        User creates their own profile.
        Used by Google onboarding.
      */
      allow create: if isSignedIn()
        && request.auth.uid == userId;


      /*
        Developer can change verification fields only.
      */
      allow update: if isDeveloper()
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "verified",
            "verifiedAt",
            "verifiedBy"
          ]);


      /*
        User can add/remove their own FCM token.
      */
      allow update: if isSignedIn()
        && request.auth.uid == userId
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "fcmTokens"
          ]);


      /*
        User can update their own profile.

        Protected fields:
        - verified
        - verifiedAt
        - verifiedBy
        - followers
        - following
        - fcmTokens
      */
      allow update: if isSignedIn()
        && request.auth.uid == userId
        && !request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasAny([
            "verified",
            "verifiedAt",
            "verifiedBy",
            "followers",
            "following",
            "fcmTokens"
          ]);


      /*
        Follow / Unfollow.

        The app updates:
        1. Current user's following  (own document)
        2. Target user's followers   (someone else's document)

        Each document update is validated separately.
        Exactly one entry may be added or removed per update.
      */

      // Step 1: my own "following" list
      allow update: if isSignedIn()
        && request.auth.uid == userId
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "following"
          ])
        && !(userId in request.resource.data.following)
        && (
          request.resource.data.following.size()
            == resource.data.get("following", []).size() + 1
          ||
          request.resource.data.following.size()
            == resource.data.get("following", []).size() - 1
        );


      // Step 2: the other user's "followers" list
      allow update: if isSignedIn()
        && request.auth.uid != userId
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "followers"
          ])
        && (
          (
            request.auth.uid in request.resource.data.followers
            && !(request.auth.uid in resource.data.get("followers", []))
            && request.resource.data.followers.size()
                == resource.data.get("followers", []).size() + 1
          )
          ||
          (
            request.auth.uid in resource.data.get("followers", [])
            && !(request.auth.uid in request.resource.data.followers)
            && request.resource.data.followers.size()
                == resource.data.get("followers", []).size() - 1
          )
        );


      /*
        Delete own account/profile document.
      */
      allow delete: if isSignedIn()
        && request.auth.uid == userId;
    }


    /* =====================================================
       POSTS
       ===================================================== */

    match /posts/{postId} {

      /*
        Feed is available to signed-in users.
      */
      allow read: if isSignedIn();


      /*
        Create own post.
      */
      allow create: if isSignedIn()
        && request.resource.data.authorId == request.auth.uid;


      /*
        Post owner can edit normal post fields.

        Protected interaction fields:
        - likes
        - likedBy
        - saves
        - savedBy
        - shares
        - commentCount
      */
      allow update: if isSignedIn()
        && (
          (
            resource.data.authorId == request.auth.uid
            && !request.resource.data
              .diff(resource.data)
              .affectedKeys()
              .hasAny([
                "authorId",
                "likes",
                "likedBy",
                "saves",
                "savedBy",
                "shares",
                "commentCount"
              ])
          )

          ||

          /*
            Interaction updates.
          */
          (
            request.resource.data
              .diff(resource.data)
              .affectedKeys()
              .hasOnly([
                "likes",
                "likedBy",
                "saves",
                "savedBy",
                "shares",
                "commentCount",
                "updatedAt"
              ])
          )
        );


      /*
        Delete own post.
      */
      allow delete: if isSignedIn()
        && resource.data.authorId == request.auth.uid;


      /* ===================================================
         POST COMMENTS
         =================================================== */

      match /comments/{commentId} {

        allow read: if isSignedIn();


        /*
          User can create their own comment.
        */
        allow create: if isSignedIn()
          && request.resource.data.userId == request.auth.uid;


        /*
          Comment owner can edit comment.
          userId cannot be changed.
        */
        allow update: if isSignedIn()
          && resource.data.userId == request.auth.uid
          && request.resource.data.userId == resource.data.userId;


        /*
          Delete own comment.
        */
        allow delete: if isSignedIn()
          && resource.data.userId == request.auth.uid;
      }
    }


    /* =====================================================
       STORIES
       ===================================================== */

    match /stories/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.authorId == request.auth.uid;
      allow update: if request.auth != null && (
        resource.data.authorId == request.auth.uid ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['viewedBy'])
      );
      allow delete: if request.auth != null && resource.data.authorId == request.auth.uid;
    }


    /* =====================================================
       REELS
       ===================================================== */

    match /reels/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.authorId == request.auth.uid;
      allow update: if request.auth != null && (
        resource.data.authorId == request.auth.uid ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes','likedBy','commentCount','shares'])
      );
      allow delete: if request.auth != null && resource.data.authorId == request.auth.uid;

      match /comments/{cid} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
        allow update: if request.auth != null && resource.data.userId == request.auth.uid;
        allow delete: if request.auth != null && resource.data.userId == request.auth.uid;
      }
    }


    /* =====================================================
       NOTIFICATIONS
       ===================================================== */

    match /notifications/{notificationId} {

      /*
        Only notification receiver can read.
      */
      allow read: if isSignedIn()
        && resource.data.receiverId == request.auth.uid;


      /*
        Logged-in user can create a notification
        on behalf of themselves.
      */
      allow create: if isSignedIn()
        && request.resource.data.senderId == request.auth.uid
        && request.resource.data.receiverId != request.auth.uid;


      /*
        Receiver can mark notification as read.
      */
      allow update: if isSignedIn()
        && resource.data.receiverId == request.auth.uid
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "read",
            "isRead"
          ]);


      /*
        Receiver can delete notification.
      */
      allow delete: if isSignedIn()
        && resource.data.receiverId == request.auth.uid;
    }


    /* =====================================================
       CHATS
       ===================================================== */

    match /chats/{chatId} {

      /*
        Existing chat:
        only a member can read.
      */
      allow read: if isSignedIn()
        && request.auth.uid in resource.data.users;


      /*
        Create chat:
        creator must be one of the chat users.
      */
      allow create: if isSignedIn()
        && request.auth.uid in request.resource.data.users;


      /*
        Update existing chat.

        Only existing members can update.
        The users array cannot be changed.

        This supports:
        - updatedAt
        - lastMessage
        - lastSenderId
      */
      allow update: if isSignedIn()
        && request.auth.uid in resource.data.users
        && request.resource.data.users == resource.data.users;


      /*
        Existing member can delete chat.
      */
      allow delete: if isSignedIn()
        && request.auth.uid in resource.data.users;


      /* ===================================================
         CHAT MESSAGES
         =================================================== */

      match /messages/{messageId} {

        /*
          Only members of parent chat can read messages.
        */
        allow read: if isSignedIn()
          && request.auth.uid in
            get(
              /databases/$(database)/documents/chats/$(chatId)
            ).data.users;


        /*
          Create message:
          sender must be current logged-in user
          and must belong to the chat.
        */
        allow create: if isSignedIn()
          && request.resource.data.senderId == request.auth.uid
          && request.auth.uid in
            get(
              /databases/$(database)/documents/chats/$(chatId)
            ).data.users;


        /*
          Sender can delete own message.
        */
        allow delete: if isSignedIn()
          && resource.data.senderId == request.auth.uid;


        /*
          Sender or receiver can update message.

          The app uses this to mark:
          read: true
        */
        allow update: if isSignedIn()
          && (
            resource.data.senderId == request.auth.uid
            ||
            resource.data.receiverId == request.auth.uid
          );
      }
    }


    /* =====================================================
       DEFAULT DENY
       ===================================================== */

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
